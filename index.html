Cấu Trúc Cây Merkle (Merkle Tree Proof)

Cryptographic Integrity

Cây Merkle gom nhóm tất cả mã băm tài liệu (Leaf Nodes) trong sổ cái thành một Merkle Root duy nhất lưu trong Block Header. Bất kỳ thay đổi nào ở một tài liệu lá sẽ làm thay đổi toàn bộ nhánh lên tới Merkle Root.

Tối Ưu Chi Phí Lưu Trữ (Off-chain vs On-chain)

Không lưu trực tiếp tệp PDF/DOCX nặng lên Blockchain để tránh phí Gas cao. Chỉ lưu chuỗi băm bytes32 docHash (SHA-256) cùng Metadata.

Độ Phức Tạp Tra Cứu O(1)

Sử dụng cấu trúc mapping(bytes32 => Record) giúp việc kiểm tra tài liệu đã công chứng hay chưa diễn ra tức thì mà không cần duyệt vòng lặp.

Sự Kiện Truy Vết (Event Logs)

Phát sự kiện event DocumentNotarized với từ khóa indexed để ứng dụng Web3 Frontend lắng nghe và lọc lịch sử dễ dàng.

// SPDX-License-Identifier: MIT
pragma solidity ^0.8.20;

/**
 * @title BlockNotary - Hệ Thống Công Chứng Tài Liệu Số Bằng Blockchain
 * @notice Lưu trữ dấu vân tay mật mã học SHA-256 của tài liệu và xác thực tính toàn vẹn
 */
contract DocumentNotary {

    // Cấu trúc lưu trữ thông tin một bản ghi công chứng
    struct Record {
        bytes32 docHash;        // Mã băm SHA-256 của tài liệu (32 bytes)
        string fileName;        // Tên tài liệu gốc
        string category;        // Phân loại tài liệu (Hợp đồng, Văn bằng...)
        string ownerName;       // Tên chủ sở hữu / Cơ quan công chứng
        string note;            // Ghi chú đính kèm
        address notarizedBy;    // Địa chỉ ví Web3 thực hiện ký công chứng
        uint256 timestamp;      // Thời gian đóng dấu khối (Unix Block Timestamp)
        uint256 blockNumber;    // Số thứ tự Block ghi nhận
        bool isRegistered;      // Cờ đánh dấu tồn tại
    }

    // Mapping từ mã băm SHA-256 sang bản ghi chi tiết (Tra cứu O(1))
    mapping(bytes32 => Record) private records;

    // Danh sách tất cả các mã băm đã công chứng để thống kê Sổ cái
    bytes32[] public allDocHashes;

    // Sự kiện phát ra khi một tài liệu mới được công chứng thành công
    event DocumentNotarized(
        bytes32 indexed docHash,
        address indexed notarizedBy,
        string fileName,
        string ownerName,
        uint256 timestamp,
        uint256 blockNumber
    );

    // Lỗi tùy chỉnh tiết kiệm Gas
    error DocumentAlreadyNotarized(bytes32 docHash, uint256 existingTimestamp);
    error InvalidDocumentHash();

    /**
     * @dev Ghi nhận mã băm tài liệu mới lên chuỗi khối
     * @param _docHash Mã băm SHA-256 của tài liệu
     * @param _fileName Tên tệp gốc
     * @param _category Phân loại hồ sơ
     * @param _ownerName Tên người sở hữu
     * @param _note Ghi chú bổ sung
     */
    function notarizeDocument(
        bytes32 _docHash,
        string calldata _fileName,
        string calldata _category,
        string calldata _ownerName,
        string calldata _note
    ) external {
        if (_docHash == bytes32(0)) {
            revert InvalidDocumentHash();
        }
        // Đảm bảo một tài liệu không bị ghi đè mốc thời gian ban đầu
        if (records[_docHash].isRegistered) {
            revert DocumentAlreadyNotarized(_docHash, records[_docHash].timestamp);
        }

        records[_docHash] = Record({
            docHash: _docHash,
            fileName: _fileName,
            category: _category,
            ownerName: _ownerName,
            note: _note,
            notarizedBy: msg.sender,
            timestamp: block.timestamp,
            blockNumber: block.number,
            isRegistered: true
        });

        allDocHashes.push(_docHash);

        emit DocumentNotarized(
            _docHash,
            msg.sender,
            _fileName,
            _ownerName,
            block.timestamp,
            block.number
        );
    }

    /**
     * @dev Xác thực tính toàn vẹn của một tài liệu dựa trên mã băm SHA-256
     * @param _docHash Mã băm SHA-256 cần kiểm tra
     * @return isVerified Trạng thái tồn tại và nguyên vẹn
     * @return record Chi tiết bản ghi trên Blockchain
     */
    function verifyDocument(bytes32 _docHash)
        external
        view
        returns (bool isVerified, Record memory record)
    {
        Record memory rec = records[_docHash];
        return (rec.isRegistered, rec);
    }

    /**
     * @dev Trả về tổng số tài liệu đã được công chứng trên Smart Contract
     */
    function getTotalNotarizedDocuments() external view returns (uint256) {
        return allDocHashes.length;
    }
}
